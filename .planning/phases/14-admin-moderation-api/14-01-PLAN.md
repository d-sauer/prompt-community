```typescript
// Worker test pattern (vitest.workers.config.ts / cloudflare:test)
import { describe, it, expect, beforeAll } from 'vitest'
import { env, createExecutionContext, waitOnExecutionContext } from 'cloudflare:test'
import { sign } from 'hono/jwt'
import app from '../index'

async function request(req: Request): Promise<Response> {
  const ctx = createExecutionContext()
  const res = await app.fetch(req, env, ctx)
  await waitOnExecutionContext(ctx)
  return res
}

// Seeded IDs from scripts/seed.sql
const SEEDED_MAINTAINER_ID = '01DEVMAINT00000000000000001'
const SEEDED_USER_ID       = '01DEVUSER000000000000000001'

// JWT minting pattern
const jwtSecret = (env as Record<string, string>).JWT_SECRET ?? 'test-secret'
maintainerToken = await sign({ sub: SEEDED_MAINTAINER_ID, role: 'maintainer', login: 'dev-maintainer', exp: ... }, jwtSecret)
userToken       = await sign({ sub: SEEDED_USER_ID, role: 'user', login: 'dev-user', exp: ... }, jwtSecret)

// Admin spec: also insert a flagged prompt in beforeAll (seed.sql has none)
FLAGGED_PROMPT_ID = '01PFLAGGED0000000000000001'
await env.DB.prepare(`INSERT OR IGNORE INTO prompts (id, author_id, title, body, status) VALUES (?, ?, ?, ?, 'flagged')`)
  .bind(FLAGGED_PROMPT_ID, SEEDED_USER_ID, 'Suspicious Prompt', 'Bad content').run()

// Error shape per API-27 — both fields required
expect(body).toHaveProperty('error')
expect(body).toHaveProperty('code')
```
