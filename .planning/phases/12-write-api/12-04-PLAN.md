```typescript
export const comments = sqliteTable('comments', {
  id:         text('id').primaryKey().$defaultFn(() => ulid()),

export const reactions = sqliteTable('reactions', {
  prompt_id:  text('prompt_id').notNull().references(() => prompts.id, { onDelete: 'cascade' }),
  user_id:    text('user_id').notNull().references(() => users.id, { onDelete: 'cascade' }),
  emoji:      text('emoji', { enum: ['thumbs_up', 'heart', 'rocket'] }).notNull(),
  created_at: text('created_at').notNull().default(tsNow),
}, (t) => [
  primaryKey({ columns: [t.prompt_id, t.user_id, t.emoji] }),
  // Composite PK: one reaction per (prompt, user, emoji)
])

export const notifications = sqliteTable('notifications', {
  id:         text('id').primaryKey().$defaultFn(() => ulid()),
  user_id:    text('user_id').notNull().references(() => users.id),
  type:       text('type').notNull(),    // 'comment_added' | 'reaction_added'
  prompt_id:  text('prompt_id').references(() => prompts.id),
  comment_id: text('comment_id').references(() => comments.id),
  read_at:    text('read_at'),
  created_at: text('created_at').notNull().default(tsNow),
})
```

Notification trigger rules (from 12-CONTEXT.md):
- Skip if actor.id === prompt.author_id (no self-notifications)
- comment_added: populate comment_id
- reaction_added: comment_id = null

Soft delete pattern for comments (returns '[deleted]' in GET /comments read handler already):
```typescript
await db.update(schema.comments)
  .set({ deleted_at: sql`(strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))` })
  .where(and(eq(schema.comments.id, commentId)))
```

Reaction count aggregation (same pattern as GET /prompts/:id):
```typescript
const reactionRows = await db.select().from(schema.reactions).where(eq(schema.reactions.prompt_id, id))
const reaction_counts = { thumbs_up: 0, heart: 0, rocket: 0 }
for (const r of reactionRows) {
  if (r.emoji === 'thumbs_up') reaction_counts.thumbs_up++
  else if (r.emoji === 'heart') reaction_counts.heart++
  else if (r.emoji === 'rocket') reaction_counts.rocket++
}
```

409 Conflict for duplicate reaction — D1 will throw a unique constraint error on composite PK insert:
- Catch the error, check if it's a UNIQUE constraint violation, return 409
- Pattern: `catch (e: unknown) { if (String(e).includes('UNIQUE') || String(e).includes('SQLITE_CONSTRAINT')) { return c.json({ error: 'already_reacted', code: 'already_reacted' }, 409) } throw e }`
